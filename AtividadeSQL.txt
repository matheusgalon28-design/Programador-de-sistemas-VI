CREATE TABLE evento (
	
	codigo_evento SERIAL PRIMARY KEY,
    nome VARCHAR(150) NOT NULL,
    descricao TEXT,
    data_realizacao DATE NOT NULL,
    carga_horaria INTEGER NOT NULL,
    codigo_departamento INTEGER NOT NULL,
    id_espaco INTEGER NOT NULL,

    	CONSTRAINT fk_evento_departamento
        FOREIGN KEY (codigo_departamento)
        REFERENCES departamento(codigo_departamento),

    	CONSTRAINT fk_evento_espaco
        FOREIGN KEY (id_espaco)
        REFERENCES espaco(id_espaco)
);

	CREATE TABLE palestrante (
    id_palestrante SERIAL PRIMARY KEY,
    nome_completo VARCHAR(150) NOT NULL,
    email VARCHAR(150) NOT NULL,
    telefone VARCHAR(20),
    instituicao_origem VARCHAR(150)
);



	CREATE TABLE participacao_palestrante (
    id_participacao SERIAL PRIMARY KEY,

    id_palestrante INTEGER NOT NULL,
    codigo_evento INTEGER NOT NULL,

    	CONSTRAINT fk_participacao_palestrante
        FOREIGN KEY (id_palestrante)
        REFERENCES palestrante(id_palestrante),

   	 	CONSTRAINT fk_participacao_evento
        FOREIGN KEY (codigo_evento)
        REFERENCES evento(codigo_evento),

    	CONSTRAINT uk_palestrante_evento
        UNIQUE (id_palestrante, codigo_evento)
);




	CREATE TABLE estudante (
    matricula VARCHAR(20) PRIMARY KEY,
    nome_completo VARCHAR(150) NOT NULL,
    curso VARCHAR(100) NOT NULL,
    periodo INTEGER NOT NULL,
    email_institucional VARCHAR(150) NOT NULL
);



	CREATE TABLE inscricao (
    id_inscricao SERIAL PRIMARY KEY,

    data_inscricao DATE NOT NULL,

    situacao VARCHAR(30) NOT NULL
        CHECK (
            situacao IN (
                'INSCRITO',
                'CONFIRMADO',
                'CANCELADO',
                'CERTIFICADO'
            )
        ),

    matricula VARCHAR(20) NOT NULL,
    codigo_evento INTEGER NOT NULL,

    	CONSTRAINT fk_inscricao_estudante
        FOREIGN KEY (matricula)
        REFERENCES estudante(matricula),

    	CONSTRAINT fk_inscricao_evento
        FOREIGN KEY (codigo_evento)
        REFERENCES evento(codigo_evento),

    	CONSTRAINT uk_estudante_evento
        UNIQUE (matricula, codigo_evento)
);

	
		CREATE TABLE certificado (

	    data_emissao DATE NOT NULL,

	    id_inscricao INTEGER UNIQUE NOT NULL,
	
	    	CONSTRAINT fk_certificado_inscricao

	        REFERENCES inscricao(id_inscricao)

CREATE TABLE patrocinador (
    cnpj VARCHAR(18) PRIMARY KEY,
    nome VARCHAR(150) NOT NULL,
    telefone VARCHAR(20),
    cidade_origem VARCHAR(100)
);


CREATE TABLE patrocinio (
    id_patrocinio SERIAL PRIMARY KEY,

    cnpj VARCHAR(18) NOT NULL,
    codigo_evento INTEGER NOT NULL,

    CONSTRAINT fk_patrocinio_patrocinador
        FOREIGN KEY (cnpj)
        REFERENCES patrocinador(cnpj),

    CONSTRAINT fk_patrocinio_evento
        FOREIGN KEY (codigo_evento)
        REFERENCES evento(codigo_evento),

    CONSTRAINT uk_patrocinador_evento
        UNIQUE (cnpj, codigo_evento)
);
